# 各城市销售额 import matplotlib.pyplot as plt cities = ['北京','上海','广州','深圳','杭州'] sales = [986420, 836210, 672340, 548120, 396420] plt.barh(cities, sales) plt.xlabel('销售额（元）') plt.show()
